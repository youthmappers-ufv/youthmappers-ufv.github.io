membros = FileAttachment("dados/painel/membros.json").json()
periodos = FileAttachment("dados/painel/periodos.json").json()

osmMensal = FileAttachment("dados/painel/osm-mensal.json").json()

// Só ações que já aconteceram entram nas estatísticas

cores = Object.freeze({vinho: "#7a1f21", marinho: "#324170", ouro: "#bfa13c", cinza: "#5f6170", nevoa: "#b9c2dc"})
// Uma cor por tipo (na ordem do vocabulário); acrescente cores se surgirem tipos novos
paleta = [cores.vinho, cores.marinho, cores.ouro, cores.cinza, cores.nevoa, "#d9c38a", "#8c4a4c", "#7f8bb3"]
fmt = (n, casas = 0) => (n ?? 0).toLocaleString("pt-BR", {maximumFractionDigits: casas})
inteiro = (d) => Number.isInteger(d) ? d.toLocaleString("pt-BR") : ""
// Mensagem exibida no lugar de um gráfico sem dados
semDados = (msg = "Sem registros para este período.") => html`<div class="painel-vazio">${msg}</div>`
// Anos presentes nos dados: as opções do filtro crescem sozinhas com o tempo
anos = Array.from(new Set([
  ...eventos.map(e => e.data.slice(0, 4)),
  ...publicacoes.map(p => String(p.ano)),
  ...projetos.map(p => p.inicio.slice(0, 4)),
  ...osmMensal.map(m => m.mes.slice(0, 4))
])).sort().reverse()
semestreAtual = meta.semestre_atual
// Semestres desde o primeiro período registrado até o atual
semestres = {
  const primeiro = d3.min(periodos, p => p.inicio)
  if (!primeiro) return []
  const lista = []
  for (let a = +primeiro.slice(0, 4); a <= +semestreAtual.slice(0, 4); a++)
    for (const s of [1, 2]) { const id = `${a}.${s}`; if (id <= semestreAtual) lista.push(id) }
  return lista
}
limites = (s) => { const [a, n] = s.split("."); return n === "1" ? [`${a}-01`, `${a}-07`] : [`${a}-08`, `${a}-12`] }
// Um período está ativo no semestre se começou antes do fim e terminou depois do início.
// Quem mudou de função no semestre (ex.: voluntário que virou bolsista) conta uma
// única vez, com a função mais recente.
ativosPorSemestre = semestres.flatMap(s => {
  const [ini, fim] = limites(s)
  const ativos = periodos.filter(p => p.inicio <= fim && (p.fim === null || p.fim >= ini))
  return d3.groups(ativos, p => p.membro).map(([membro, ps]) =>
    ({semestre: s, membro, funcao: d3.greatest(ps, p => p.inicio).funcao}))
})
equipeAtual = ativosPorSemestre.filter(d => d.semestre === semestreAtual && d.funcao !== "coordenacao")
ativoNoAno = (inicio, fim, a) => inicio <= `${a}-12` && (fim === null || fim >= `${a}-01`)
eventosF = ano === "Todos" ? eventos : eventos.filter(e => e.data.startsWith(ano))
projetosF = ano === "Todos" ? projetos : projetos.filter(p => ativoNoAno(p.inicio, p.fim, ano))
publicacoesF = ano === "Todos" ? publicacoes : publicacoes.filter(p => String(p.ano) === ano)
osmF = ano === "Todos" ? osmMensal : osmMensal.filter(m => m.mes.startsWith(ano))
osmAcumulado = { let a = 0; return osmMensal.map(m => ({data: new Date(`${m.mes}-15T00:00:00Z`), acumulado: a += m.edicoes})) }
// Faixa dourada que destaca o ano escolhido nas linhas do tempo
destaqueAno = ano === "Todos" ? [] :
  [Plot.rectX([{x1: new Date(`${ano}-01-01T00:00:00Z`), x2: new Date(`${+ano + 1}-01-01T00:00:00Z`)}],
              {x1: "x1", x2: "x2", fill: cores.ouro, fillOpacity: 0.15})]
// Eixo mensal: só o ano em "Todos"; iniciais dos meses quando um ano está selecionado
rotuloMes = (d) => ano === "Todos" ? (d.endsWith("-01") ? d.slice(0, 4) : "") : "JFMAMJJASOND"[+d.slice(5) - 1]
projetosPorAno = anos.slice().sort().flatMap(a =>
  projetos.filter(p => ativoNoAno(p.inicio, p.fim, a)).map(p => ({ano: a, tipo: p.tipo})))
publicoAno = d3.rollups(eventosF.filter(e => e.publico !== null), v => [
    {tipo: "Comunidade externa", n: d3.sum(v, e => e.publico_externo ?? 0)},
    {tipo: "Comunidade da UFV", n: d3.sum(v, e => e.publico - (e.publico_externo ?? 0))}
  ], e => e.data.slice(0, 4)).flatMap(([a, xs]) => xs.map(x => ({ano: a, ...x})))
// Indicador do OSM: saldo em destaque e, abaixo, criados/modificados/removidos
indicadorOsm = (campo, casas = 0, feminino = false) => {
  const soma = (c) => d3.sum(osmF, m => m[c] ?? 0)
  const g = feminino ? "a" : "o"
  return html`<div>${fmt(soma(campo), casas)}</div>
    <div class="vb-detalhe">${fmt(soma(campo + "_criados"), casas)} criad${g}s ·
      ${fmt(soma(campo + "_modificados"))} modificad${g}s ·
      ${fmt(soma(campo + "_removidos"), casas)} removid${g}s</div>`
}

Como ler os números do mapeamento

Edificações, vias, pontos de interesse e cursos d’água são saldos: o que foi criado menos o que foi removido (para vias e cursos d’água, também alongamentos menos encurtamentos). Por isso um mês pode ficar negativo: acontece em meses de validação, quando o trabalho é corrigir mapeamentos anteriores. As modificações não entram no saldo e aparecem à parte nos indicadores. Vias incluem tudo o que é mapeado como highway, inclusive calçadas e trilhas; cursos d’água são rios, canais, córregos e valas. Barras mais claras indicam o mês em andamento. Definições completas na ajuda do painel ohsomeNow.
